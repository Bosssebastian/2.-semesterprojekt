// for the sender (sending pico)

#include "Uart.h"

int main() {
    Uart uart0(0, 0, 1, 115200);  // UART0, pins GP0/GP1
    uart0.init();

    uart0.send('A');
}

//for the reciever (recieving pico)

#include "Uart.h"

int main() {
    Uart uart1(1, 4, 5, 115200);  // UART1, different pins
    uart1.init();

    char c = uart1.receive();
}
